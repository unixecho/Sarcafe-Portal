/** A time range such as 07:00–14:00 inside Hebrew is flipped by the bidi rules; it is set apart left-to-right (the app does the same, AlertText). */
const RANGE = /(\d{1,2}[:/]\d{1,2}[–-]\d{1,2}[:/]\d{1,2})/
const Isolate = ({ text }: { text: string }) => <>{text.split(RANGE).map((p, i) => (i % 2 ? <bdi key={i} dir="ltr">{p}</bdi> : p))}</>

      <div className="short">

          <button key={e.id} type="button" className="surf short__tile" onClick={() => onOpen(e)}>

        ) : reqs.length === 0 ? <p className="oneline">אין בקשות פתוחות.</p> : reqs.map((r) => (

            <Button destructive icon={<X size={16} />} aria-label={`דחיית הבקשה של ${r.who}`} onClick={() => { setReqs((x) => x.filter((y) => y.id !== r.id)); toast(`הבקשה של ${r.who} נדחתה`) }} />
          </div>
        ))}
      </section>
      <section className="surf card" aria-labelledby="week-t">
        <h2 className="card__title" id="week-t">השבוע ב{data.name}</h2>

            <div className="week__row" role="row" key={d.index}>
              <span role="rowheader"><b>{d.short}</b> <small dir="ltr">{d.dm}</small></span>
              {(['morning', 'evening'] as const).map((p) => { const r = rosterFor(branch, d, p); return <span role="cell" key={p} className={r.length ? '' : 'week__none'}>{r.length ? r.join(', ') : 'סגור'}</span> })}
            </div>
          ))}
        </div>
      </section>
    </>
  )
})

const StaffView = memo(function StaffView({ branch, data, role, toast }: { branch: BranchId; data: BranchData; role: Role; toast: (m: string) => void }) {
  const days = useMemo(() => nextDays(7), [])
  const people = NAMES.slice(0, data.staffCount)
  const status = (n: string) => {
    if (data.onNow.includes(n)) return 'במשמרת עכשיו'
    const d = days.find((x) => x.index > 0 && (['morning', 'evening'] as const).some((p) => rosterFor(branch, x, p).includes(n)))
    if (!d) return 'אין משמרת קרובה'
    const p = (['morning', 'evening'] as const).find((q) => rosterFor(branch, d, q).includes(n))!
    return `הבאה: ${d.short} ב${PERIODS[p].label}`
  }
  return (
    <section className="surf card" aria-labelledby="staff-t">
      <h2 className="card__title" id="staff-t"><Users size={17} aria-hidden="true" />הצוות ב{data.name} · {people.length}</h2>
      <ul className="people people--list">
        {people.map((n) => <li key={n}><span className="avatar" aria-hidden="true">{n[0]}</span><span className="row__copy"><strong>{n}</strong><small>{status(n)}</small></span>{data.onNow.includes(n) && <Pill quiet>עכשיו</Pill>}</li>)}
      </ul>
      {role !== 'staff' && <div className="card__actions"><Button variant="filled" icon={<Users size={16} />} onClick={() => toast('קישור ההצטרפות הועתק')}>קישור הצטרפות לעובד חדש</Button></div>}
    </section>
  )
})

// ─────────────────────────────────────────────── app
export default function App() {
  const [tab, setTab] = useState('home')
  const [branchUi, setBranchUi] = useState<BranchId>('givat-haviva')
  const [branch, setBranch] = useState<BranchId>('givat-haviva')
  const [, startTransition] = useTransition()
  const [lens, setLens] = useState('manager')
  const [look, setLook] = useState('glass')
  const [push, setPush] = useState(true)
  const [shiftAlerts, setShiftAlerts] = useState(true)
  const [idle, setIdle] = useState(60)
  const [published, setPublished] = useState<Record<string, boolean>>({})
  const [read, setRead] = useState<Record<string, boolean>>({})
  const [toastMsg, setToastMsg] = useState<{ k: number; m: string } | null>(null)
  const role = LENS_ROLE[lens]!
  const data = BRANCHES[branch]
  const hour = useMemo(jerusalemHour, [])

  // usage history (per role, kept on the device); the preview starts from example usage
  const [usageState, setUsageState] = useState<{ usage: Usage; sample: boolean }>(() => loadUsage('owner'))
  useEffect(() => { setUsageState(loadUsage(role)) }, [role])
  const ctx: Context = useMemo(() => ({ role, hour, usage: usageState.usage }), [role, hour, usageState.usage])
  const badges = useMemo(() => pendingBadges(data, role), [data, role])

  // glisse's refraction filter (Chromium only) is re-run every frame for glass that sits over scrolling content, which is what
  // made scrolling stutter. While the page scrolls the lens is swapped for plain blur; it comes back 180 ms after it stops.
  useEffect(() => {
    const root = document.documentElement; let timer = 0
    const onScroll = () => { root.classList.add('is-scrolling'); window.clearTimeout(timer); timer = window.setTimeout(() => root.classList.remove('is-scrolling'), 180) }
    window.addEventListener('scroll', onScroll, { passive: true })
    return () => { window.removeEventListener('scroll', onScroll); window.clearTimeout(timer); root.classList.remove('is-scrolling') }
  }, [])

  // A switch (branch or tab) animates glisse's moving thumb. For the next 0.9 s the glass AROUND it uses plain blur; the thumb keeps its lens.
  const switching = useRef(0)
  const settle = useCallback(() => {
    const root = document.documentElement; root.classList.add('is-switching'); window.clearTimeout(switching.current)
    switching.current = window.setTimeout(() => root.classList.remove('is-switching'), 900)
  }, [])

  const toast = useCallback((m: string) => setToastMsg({ k: Date.now(), m }), [])
  useEffect(() => { if (!toastMsg) return; const t = setTimeout(() => setToastMsg(null), 2800); return () => clearTimeout(t) }, [toastMsg])

  const goTab = useCallback((v: string) => { settle(); setTab(v); window.scrollTo({ top: 0, behavior: 'instant' as ScrollBehavior }) }, [settle])
  const scrollToId = useCallback((id: string, focus?: string) => {
    window.setTimeout(() => { document.getElementById(id)?.scrollIntoView({ block: 'center', behavior: reduced() ? 'auto' : 'smooth' }); if (focus) window.setTimeout(() => document.getElementById(focus)?.focus({ preventScroll: true }), 350) }, 80)
  }, [])
  const openPath = useCallback((href: string) => {
    const hit = TAB_OF.find(([re]) => re.test(href))
    if (hit) { goTab(hit[1]); return }
    toast(`בגרסה האמיתית נפתח: ${href}`)
  }, [goTab, toast])

  const openEntry = useCallback((e: Entry) => {
    setUsageState((s) => ({ usage: recordUse(role, s.usage, e.id), sample: false }))
    switch (e.action) {
      case 'compose': goTab('home'); scrollToId('compose', 'compose-text'); return
      case 'publish': setPublished((p) => ({ ...p, [branch]: true })); toast('השינויים בתפריט פורסמו'); return
      case 'look': goTab('home'); scrollToId('look'); return
      case 'push': goTab('home'); scrollToId('controls'); return
      case 'install': toast('הוספה למסך הבית (בטלפון אמיתי נפתחת הנחיית ההתקנה)'); return
      case 'signout': toast('התנתקות (בתצוגה לא מתנתקים)'); return
    }
    openPath(e.href)
  }, [role, branch, goTab, scrollToId, toast, openPath])
  const resetUsageCb = useCallback(() => { resetUsage(role); setUsageState({ usage: {}, sample: false }); saveUsage(role, {}); toast('היסטוריית השימוש נמחקה') }, [role, toast])

  const onBranch = useCallback((v: string) => { settle(); setBranchUi(v as BranchId); startTransition(() => setBranch(v as BranchId)) }, [settle])

  // notifications: the same morphing glass menu as the account button
  const notices = useMemo(() => NOTICES.filter((n) => n.role.includes(role)), [role])
  const unreadOf = (id: string, initial: boolean) => (read[id] ? false : initial)
  const unread = notices.filter((n) => unreadOf(n.id, n.unread)).length
  const noticeItems: MenuEntry[] = useMemo(() => {
    const items: MenuEntry[] = [{
      label: <span className="nt nt--all"><strong>{unread ? 'סימון הכול כנקרא' : 'הכול נקרא'}</strong><small>{unread ? `${unread} התראות חדשות` : 'אין התראות חדשות'}</small></span>,
      icon: <Check size={16} />, disabled: unread === 0, onSelect: () => { setRead(Object.fromEntries(notices.map((n) => [n.id, true]))); toast('סומנו כנקראו') },
    }, 'separator']
    for (const n of notices) {
      const u = read[n.id] ? false : n.unread
      items.push({
        icon: <Bell size={16} />,
        label: <span className="nt"><span className="nt__top"><strong>{n.title}</strong>{u && <span className="pill pill--tiny">חדש</span>}</span><small><Isolate text={n.body} /></small><small className="nt__time">{n.time}</small></span>,
        onSelect: () => { setRead((r) => ({ ...r, [n.id]: true })); openPath(n.path) },
      })
    }
    return items
  }, [notices, read, unread, openPath, toast])
  const accountItems: MenuEntry[] = useMemo(() => [
    { label: 'הפרופיל שלי', icon: <UserRound size={18} />, onSelect: () => toast('בגרסה האמיתית נפתח: /staff/profile') },
    { label: 'מראה האפליקציה', icon: <Palette size={18} />, onSelect: () => { goTab('home'); scrollToId('look') } },
    { label: 'התקנה על מסך הבית', icon: <Download size={18} />, onSelect: () => toast('הוספה למסך הבית') },
    'separator',
    { label: 'התנתקות', icon: <LogOut size={18} />, destructive: true, onSelect: () => toast('התנתקות (בתצוגה לא מתנתקים)') },
  ], [goTab, scrollToId, toast])

  const groups = useMemo(() => HOME_GROUPS[role].map((g) => ({ title: g.title, entries: g.ids.map((id) => INDEX.find((e) => e.id === id)!).filter(Boolean) })), [role])
  const allGroups = useMemo(() => {
    const m = new Map<string, Entry[]>()
    for (const e of INDEX) if (visibleTo(e, role)) m.set(e.group, [...(m.get(e.group) ?? []), e])
    return [...m].map(([title, entries]) => ({ title, entries }))
  }, [role])

  const date = useMemo(() => new Intl.DateTimeFormat('he-IL', { timeZone: 'Asia/Jerusalem', weekday: 'long', day: 'numeric', month: 'long' }).format(new Date()), [])
  const manager = role !== 'staff'
  const tomorrow = useMemo(() => { const d = nextDays(2)[1]!; return { d, r: rosterFor(branch, d, 'morning') } }, [branch])

  // Hebrew reads right to left, so the first tab sits on the right: the bar is an LTR box with the items reversed.
  const tabItems = useMemo(() => [
    { value: 'more', label: 'עוד', icon: <MoreHorizontal size={22} /> },
    { value: 'pos', label: 'קופה', icon: <Receipt size={22} /> },
    { value: 'staff', label: 'צוות', icon: <Users size={22} /> },
    { value: 'schedule', label: 'סידור', icon: <CalendarClock size={22} /> },
    { value: 'home', label: 'בית', icon: <House size={22} /> },
  ], [])

  return (
    <>
      <div className="bg" aria-hidden="true" style={{ backgroundImage: `linear-gradient(180deg, rgb(21 15 12 / .5), rgb(21 15 12 / .68) 45%, rgb(21 15 12 / .88)), url(${BACKDROP})` }} />
      <div className="app" dir="rtl" lang="he">
        <TopBar branch={branchUi} onBranch={onBranch} unread={unread} items={noticeItems} accountItems={accountItems} />

        <main className="page" id="main">
          <section className="welcome">
            <div>
              <h1>שלום, Sarcafe</h1>
              <p>{role === 'staff' ? 'כל מה שצריך למשמרת שלך.' : 'כל מה שצריך כדי לנהל את היום שלך.'}</p>
            </div>
            <span className="welcome__date">{date}</span>
          </section>

          {/* HOME */}
          <div hidden={tab !== 'home'} className="view">
            <div className="lensbox"><Seg label="תצוגה" value={lens} onChange={setLens} items={LENSES} /></div>
            <Shortcuts ctx={ctx} badges={badges} onOpen={openEntry} />
            {manager ? (
              <>
                <BranchPanels data={data} published={!!published[branch]} onPublish={() => { setPublished((p) => ({ ...p, [branch]: true })); toast('השינויים בתפריט פורסמו') }} onGo={openPath} />
                <QuickMessage branch={branch} data={data} onSent={toast} />
              </>
            ) : (
              <>
                <button type="button" className="surf att att--main" onClick={() => openPath('/staff/schedule')}>
                  <span className="ico"><UsersRound size={24} /></span>
                  <span className="att__copy"><strong>מי במשמרת &amp; משמרות</strong><small>מי עובד עכשיו בכל סניף, והלוח והאילוצים שלכם</small></span>
                  <ChevronLeft size={18} className="chev" aria-hidden="true" />
                </button>
                <section className="surf card" aria-labelledby="mine-t">
                  <h2 className="card__title" id="mine-t"><CalendarClock size={17} aria-hidden="true" />המשמרת הבאה שלך</h2>
                  <div className="row"><div className="row__copy"><strong>{tomorrow.d.short} · <bdi dir="ltr">{PERIODS.morning.hours}</bdi></strong><small>{data.name}{tomorrow.r.length ? ` · עם ${tomorrow.r.slice(0, 2).join(' ו')}` : ''}</small></div><Button onClick={() => openPath('/staff/schedule?tab=requests')}>בקשת החלפה</Button></div>
                  <div className="row"><div className="row__copy"><strong>צ׳קליסט פתיחה</strong><small>0 מתוך 12 הושלמו</small></div><Button variant="filled" onClick={() => toast('בגרסה האמיתית נפתח: /staff/checklists')}>להתחיל</Button></div>
                </section>
              </>
            )}

            <section className="surf card" id="controls" aria-labelledby="ctl-t">
              <h2 className="card__title" id="ctl-t">התראות{manager ? ' ושליטה מהירה' : ''}</h2>
              <div className="row"><div className="row__copy"><strong>התראות לטלפון</strong><small>משמרות, חוסרים והחלפות, גם כשהאפליקציה סגורה</small></div><Switch aria-label="התראות לטלפון" checked={push} onCheckedChange={setPush} /></div>
              <div className="row"><div className="row__copy"><strong>תזכורת משמרת הבאה ב־19:00</strong><small>הודעה עם המשמרת הבאה שלך</small></div><Switch aria-label="תזכורת משמרת הבאה" checked={shiftAlerts} onCheckedChange={setShiftAlerts} /></div>
              {manager && <div className="row row--stack"><div className="row__copy"><strong>חזרה למסך הפתיחה בקיוסק</strong><small>אחרי {idle} שניות ללא נגיעה</small></div><div dir="ltr" className="sl"><Slider aria-label="שניות ללא נגיעה בקיוסק" min={20} max={180} step={10} value={idle} onValueChange={setIdle} /></div></div>}
            </section>

            <section className="tools" aria-label="כלים ופעולות">
              <ToolSearch ctx={ctx} sample={usageState.sample} badges={badges} onPick={openEntry} onUsageReset={resetUsageCb} />
              <ToolGroups groups={groups} badges={badges} onOpen={openEntry} />
            </section>

            <section className="surf card" id="look" aria-labelledby="look-t">
              <h2 className="card__title" id="look-t"><Palette size={17} aria-hidden="true" />מראה האפליקציה</h2>
              <Seg fill label="מראה האפליקציה" value={look} onChange={setLook} items={LOOKS} />
            </section>
          </div>

          <div hidden={tab !== 'schedule'} className="view"><ScheduleView branch={branch} data={data} role={role} toast={toast} /></div>
          <div hidden={tab !== 'staff'} className="view"><StaffView branch={branch} data={data} role={role} toast={toast} /></div>
          <div hidden={tab !== 'pos'} className="view"><PosView data={data} role={role} onGo={openPath} /></div>
          <div hidden={tab !== 'more'} className="view">
            <section className="tools" aria-label="כל הכלים">
              <ToolSearch ctx={ctx} sample={usageState.sample} badges={badges} onPick={openEntry} onUsageReset={resetUsageCb} placeholder="חיפוש בכל הכלים והפעולות…" />
              <p className="note note--start">{allGroups.reduce((n, g) => n + g.entries.length, 0)} כלים ופעולות · חיפוש לפי שם, מילה נרדפת או שגיאת כתיב</p>
              <ToolGroups groups={allGroups} badges={badges} onOpen={openEntry} />
            </section>
          </div>

          <p className="note">תצוגה מקדימה בלבד · נתוני דוגמה · הקוד המקורי לא שונה</p>
        </main>

        <nav className="dock" aria-label="ניווט ראשי">
          <div dir="ltr"><TabBar aria-label="ניווט ראשי" value={tab} onValueChange={goTab} items={tabItems} /></div>
        </nav>

        <div className="toast" role="status" aria-live="polite">{toastMsg && <span key={toastMsg.k} className="toast__msg">{toastMsg.m}</span>}</div>
      </div>
    </>
  )
}
