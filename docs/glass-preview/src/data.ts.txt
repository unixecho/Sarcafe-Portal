/** The small pill next to a tool or search result when something is waiting there (the style of the "2 בקשות" badge). */
export function pendingBadges(b: BranchData, role: Role): Record<string, string> {
  const m: Record<string, string> = {}
  if (role !== 'staff') {
    if (b.requests) { m['schedule-manage'] = `${b.requests} בקשות`; m['approve-swap'] = `${b.requests} בקשות` }
    if (b.shortages) m.issues = `${b.shortages} לטיפול`
    if (b.feedback) m.feedback = `${b.feedback} חדשים`
    if (b.waitlist) m.waitlist = `${b.waitlist} חדשים`
    if (b.out.length) { m['tablet-avail'] = `${b.out.length} אזלו`; m['mark-soldout'] = `${b.out.length} אזלו` }
    if (b.unpublished) { m['publish-menu'] = 'טיוטה'; m['menu-editor'] = 'טיוטה' }
  } else {
    m['checklist-open'] = '0 מתוך 12'
    if (b.out.length) m['tablet-avail'] = `${b.out.length} אזלו`
  }
  return m
}

export type Day = { index: number; date: Date; dow: number; short: string; long: string; dm: string }
export function nextDays(count = 7): Day[] {
  const parts = new Intl.DateTimeFormat('en-CA', { timeZone: 'Asia/Jerusalem', year: 'numeric', month: '2-digit', day: '2-digit' }).format(new Date()).split('-').map(Number)
  const base = Date.UTC(parts[0]!, parts[1]! - 1, parts[2]!)
  return Array.from({ length: count }, (_, index) => {
    const date = new Date(base + index * 86400000)
    const weekday = new Intl.DateTimeFormat('he-IL', { timeZone: 'UTC', weekday: 'long' }).format(date)
    const dm = `${date.getUTCDate()}/${date.getUTCMonth() + 1}`
    return { index, date, dow: date.getUTCDay(), short: index === 0 ? 'היום' : index === 1 ? 'מחר' : weekday.replace('יום ', ''), long: weekday, dm }
  })
}

const hash = (s: string) => { let h = 2166136261; for (const c of s) { h ^= c.charCodeAt(0); h = Math.imul(h, 16777619) } return h >>> 0 }
/** Who is on a shift: sample rota, deterministic. Friday evening and Saturday are closed (nobody scheduled). */
export function rosterFor(branch: BranchId, day: Day, period: Period): string[] {
  if (day.dow === 6 || (day.dow === 5 && period === 'evening')) return []
  const seed = hash(`${branch}|${day.index}|${period}`)
  const size = branch === 'maor' ? 1 + (seed % 2) : 1 + (seed % 3) + (period === 'morning' && seed % 5 === 0 ? 1 : 0)
  const pool = NAMES.slice(0, branch === 'maor' ? 5 : 8)
  const out: string[] = []
  for (let i = 0; out.length < Math.min(size, pool.length); i++) { const n = pool[(seed + i * 3) % pool.length]!; if (!out.includes(n)) out.push(n) }
  return out
}
